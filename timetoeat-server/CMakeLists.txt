cmake_minimum_required(VERSION 3.16)
project(timetoeat_server LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(HTTP REQUIRED IMPORTED_TARGET cpp-httplib)
pkg_check_modules(DB REQUIRED IMPORTED_TARGET libmariadb)
find_package(nlohmann_json REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(CURL REQUIRED)
add_executable(timetoeat-server src/main.cpp)
target_compile_options(timetoeat-server PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(timetoeat-server PRIVATE PkgConfig::HTTP PkgConfig::DB nlohmann_json::nlohmann_json OpenSSL::Crypto CURL::libcurl)
